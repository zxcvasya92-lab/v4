name: Build Android APK

on:
  push:
    branches: [main, master]
  workflow_dispatch:

permissions:
  contents: write

env:
  GODOT_VERSION: "4.3"

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"

      - name: Install Godot and export templates
        run: |
          set -e
          BASE=https://github.com/godotengine/godot/releases/download/${GODOT_VERSION}-stable
          curl -fL -o godot.zip $BASE/Godot_v${GODOT_VERSION}-stable_linux.x86_64.zip
          curl -fL -o templates.tpz $BASE/Godot_v${GODOT_VERSION}-stable_export_templates.tpz
          unzip -q godot.zip
          sudo mv Godot_v${GODOT_VERSION}-stable_linux.x86_64 /usr/local/bin/godot
          mkdir -p ~/.local/share/godot/export_templates
          unzip -q templates.tpz
          mv templates ~/.local/share/godot/export_templates/${GODOT_VERSION}.stable

      - name: Install Android SDK parts
        run: |
          ANDROID_HOME=${ANDROID_HOME:-/usr/local/lib/android/sdk}
          SDKM=$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager
          yes | $SDKM --licenses > /dev/null || true
          $SDKM "platform-tools" "build-tools;34.0.0" "platforms;android-34" "cmake;3.10.2.4988404"
          echo "ANDROID_HOME=$ANDROID_HOME" >> $GITHUB_ENV

      - name: Debug keystore and Godot editor settings
        run: |
          keytool -keyalg RSA -genkeypair -alias androiddebugkey -keypass android -keystore $HOME/debug.keystore -storepass android -dname "CN=Android Debug,O=Android,C=US" -validity 9999 -deststoretype pkcs12
          mkdir -p ~/.config/godot
          cat > ~/.config/godot/editor_settings-${GODOT_VERSION}.tres <<EOT
          [gd_resource type="EditorSettings" format=3]

          [resource]
          export/android/android_sdk_path = "${ANDROID_HOME}"
          export/android/java_sdk_path = "${JAVA_HOME}"
          export/android/debug_keystore = "${HOME}/debug.keystore"
          export/android/debug_keystore_user = "androiddebugkey"
          export/android/debug_keystore_pass = "android"
          export/android/shutdown_adb_on_exit = true
          EOT

      - name: Check scripts (errors are shown in the log)
        continue-on-error: true
        run: godot --headless --path . --check-only --script res://main.gd

      - name: Import project
        run: godot --headless --path . --import || true

      - name: Export APK
        run: |
          mkdir -p build
          godot --headless --path . --export-debug "Android" build/BMX_Street.apk
          ls -la build
          test -s build/BMX_Street.apk

      - uses: actions/upload-artifact@v4
        with:
          name: BMX-Street-apk
          path: build/BMX_Street.apk

      - name: Publish APK to Releases (download straight on phone)
        if: github.event_name == 'push' || github.event_name == 'workflow_dispatch'
        continue-on-error: true
        uses: softprops/action-gh-release@v2
        with:
          tag_name: latest
          name: BMX Street (latest build)
          files: build/BMX_Street.apk
